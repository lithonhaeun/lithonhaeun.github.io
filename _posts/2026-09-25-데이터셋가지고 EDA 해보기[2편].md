- ### 나이 분포 보기
  ~~~
  # 나이를 연령대로 묶기 (앞으로 계속 쓸 컬럼)
   df["age_group"] = pd.cut(df["age"],
                         bins=[19, 30, 40, 50, 60, 70, 80, 200],
                         labels=["19-29", "30대", "40대", "50대", "60대", "70대", "80세+"],
                         right=False)
   # 나이 히스토그램
   fig, ax = plt.subplots(figsize=(10, 4))
   ax.hist(df["age"], bins=range(19, 101), color="#2a78d6")
   ax.axvline(df["age"].median(), color="gray", linestyle="--")   # 중앙값 위치
   ax.set_title("나이 분포 (1세 단위)")
   ax.set_xlabel("나이")
   ax.set_ylabel("인원 수")
   plt.show()
   
   # 연령대별 비율
   print((df["age_group"].value_counts(normalize=True).sort_index() * 100).round(1))
  ~~~
  pd.cut은 나이 같은 연속된 숫자를 구간으로 나눠 범주로 바꿔줍니다.
   <br>bins=range(19, 101)은 막대를 1살 단위로 하나씩 그린다는 뜻이에요.
   회색 점선은 중앙값(51세) 위치입니다~.![](/assets/img/20260925-capture-9.png)
    - **해석**
      1. 가장 많은 연령대는 50~64세입니다.
      2. 젊은 층은 적지만, 70세부터 빠르게 줄어드네요.
      3. 제가 생각한 모양은 매끄러운 곡선에 가까운 모양이었는데 지금은 거의 계단이네요.

### 인구 피라미드 보기 (성별 x 나이)
  - 아까 인구가 거의 계단 식이었으니 이 맞춰서 5세 단위로 묶어서 그려 볼게요. 
  ~~~
    # 5세 단위 연령 구간 만들기 (데이터가 19세부터라 첫 구간은 19세 하나)
    bins5 = [19] + list(range(20, 105, 5))
    labels5 = ["19"] + [f"{b}-{b+4}" for b in range(20, 100, 5)]
    df["age5"] = pd.cut(df["age"], bins=bins5, labels=labels5, right=False)
    
    # 연령 구간 × 성별 인원 수
    pyr = df.groupby(["age5", "sex"], observed=False).size().unstack(fill_value=0)
    
    # 인구 피라미드: 남자는 왼쪽(음수), 여자는 오른쪽
    fig, ax = plt.subplots(figsize=(8, 7))
    ax.barh(pyr.index.astype(str), -pyr["남자"], color="#2a78d6", label="남자")
    ax.barh(pyr.index.astype(str), pyr["여자"], color="#eb6834", label="여자")
    
    lim = pyr.values.max() * 1.1
    ax.set_xlim(-lim, lim)
    ticks = ax.get_xticks()
    ax.set_xticks(ticks)
    ax.set_xticklabels([f"{abs(int(t)):,}" for t in ticks])   # 왼쪽 음수를 양수로 표시
    ax.axvline(0, color="gray", linewidth=0.8)
    ax.legend()
    ax.set_title("인구 피라미드 (5세 단위)")
    plt.show()
    
    # 연령대별 성비 (여자 ÷ 남자)
    sr = pd.crosstab(df["age_group"], df["sex"])
    sr["여/남 비율"] = (sr["여자"] / sr["남자"]).round(2)
    print(sr)
  ~~~
![](/assets/img/20260925-capture-10.png)
- 해석
  1. 아래(20대)가 좁고, 50~64세에서 가장 넓고, 위로 갈수록 다시 좁아져요.
  2. 연령대별 여/남 비율을 계산해 보니 20~30대는 0.91로 남자가 더 많고, 50대에서 1.00으로 같아진 뒤, 70대 1.16, 80세 이상 1.86으로 여자가 크게 많아진다. 젊은 층의 남초는 1980~90년대 남아선호로 출생 성비가 남자 쪽으로 치우쳤던 영향이다.
  3. 전체 성비는 남자 49.6%, 여자 50.4%로 거의 같았다.

### 혼인상태, 학력, 주거 형태 보기
~~~
    # 막대그래프를 여러 번 그릴 거라 함수로 만들어 둠
`def plot_count(col, order=None, top=None):
    vc = df[col].value_counts()          # 값별 인원 수 (많은 순)
    if order is not None:                # 순서를 직접 정하고 싶을 때
        vc = vc.reindex(order).dropna()
    if top:                              # 상위 N개만 보고 싶을 때
        vc = vc.head(top)
    pct = vc / len(df) * 100             # 비율(%)

    fig, ax = plt.subplots(figsize=(8, 0.4 * len(vc) + 1.2))
    ax.barh(vc.index[::-1], vc.values[::-1], color="#2a78d6")
    for y, p in enumerate(pct.values[::-1]):
        ax.text(vc.values[::-1][y], y, f"  {p:.1f}%", va="center", fontsize=9)
    ax.set_xlim(0, vc.max() * 1.2)
    ax.set_title(f"{col} 분포")
    plt.show()

    print((pct.round(1)).to_string(), "\n")

    # 학력은 '낮은 → 높은' 순서가 의미 있으니 순서를 정해줌 (데이터에 있는 값만 사용)
    edu_pref = ["무학", "초등학교", "중학교", "고등학교", "2~3년제 전문대학", "4년제 대학교", "대학원"]
    edu_vals = df["education_level"].unique()
    EDU_ORDER = [e for e in edu_pref if e in edu_vals] + [e for e in edu_vals if e not in edu_pref]
    
    plot_count("marital_status")
    plot_count("education_level", order=EDU_ORDER)
    plot_count("housing_type")
~~~

- 해석
  1. 사별 8.8%는 이혼(6.3%)보다 많아요. 앞에서 본 것처럼 노년층에 여자가 훨씬 많았으니, 사별은 고령 여성에게 몰려 있을 것 같아요.
  2. 가장 많은 건 고졸(33.1%), 그다음이 4년제(27.1%)예요.무학·초졸·중졸은 고령층에, 대졸 이상은 젊은 층에 몰려 있을 것같다는 생각에

     ~~~
     # 연령대 × 학력 교차표 (각 연령대 안에서의 %)
        ct = pd.crosstab(df["age_group"], df["education_level"], normalize="index") * 100
        ct = ct[EDU_ORDER]                       # 학력 순서대로 정렬 (3단계에서 만든 순서 사용)
        
        # 학력이 높을수록 진한 파랑
        colors = ["#cde2fb", "#9ec5f4", "#6da7ec", "#3987e5", "#256abf", "#184f95", "#0d366b"]
        
        fig, ax = plt.subplots(figsize=(10, 5))
        left = [0] * len(ct)                     # 막대를 이어 붙일 시작 위치
        for i, edu in enumerate(ct.columns):
            ax.barh(ct.index.astype(str), ct[edu], left=left,
                    color=colors[i], label=edu, edgecolor="white")
            for y, (l, v) in enumerate(zip(left, ct[edu])):   # 6% 이상인 칸에만 숫자 표시
                if v >= 6:
                    ax.text(l + v / 2, y, f"{v:.0f}", ha="center", va="center", fontsize=8,
                            color="white" if i >= 3 else "black")
            left = [l + v for l, v in zip(left, ct[edu])]
        
        ax.set_xlim(0, 100)
        ax.invert_yaxis()                        # 19-29가 맨 위에 오도록
        ax.set_title("연령대별 학력 구성 (%)")
        ax.legend(ncol=4, bbox_to_anchor=(0, -0.08), loc="upper left", frameon=False, fontsize=9)
        plt.show()
        
        print(ct.round(1))
        ~~~

  2.  나이가 많을수록 초졸 이하의 비율이 커져서, 80세 이상은 69.1%가 초등학교 이하 학력이었고 무학도 28.1%에 달했다. 반대로 30대는 56.3%가 4년제 대학 이상이었다. 각 세대에서 가장 많은 학력이 80대·70대의 초졸에서 40~60대의 고졸, 20~30대의 4년제로 옮겨 가며, 한국에서 교육 기회가 세대마다 확대되어 온 흐름이 그대로 드러난다. 
  3. 대졸 이상 비율이 20대(44.7%)보다 30대(56.3%)에서 더 높다는 것이 신기하네요. 19~29세 중 상당수가 아직 대학에 재학 중이어서 최종 학력이 고졸로 집계되기 때문도 있을것 같아요.
  4. 대학원 비율 역시 학업을 마친 30~40대에서 가장 높았다.

## 역서 제가 맘에드는 인물은?
제가 고른 인물은 광주 광산구에 사는 오시자 씨예요. 오시자 씨는 광산구의 옛 골목길과 시장통에서 거친 세월을 보내며, 이웃 간의 크고 작은 다툼을 잠재우던 동네 해결사였어요. 지금은 복지관에서 어르신들의 고민을 들어주는 정신적 지주 역할을 하고 있대요.

매주 화요일이면 복지관 강당에서 징 소리와 민요 가락에 맞춰 느릿하게 춤을 추며 굳어가는 관절을 풀어요. 혼자 있는 시간에는 돋보기를 코끝에 걸치고 주현미의 노래를 틀어놓은 채 소설책을 읽고요. 한 달에 한 번은 마음 맞는 친구와 단골 중국집에서 짜장면 한 그릇을 비우는 소소한 즐거움도 있어요.

사별 후 아파트에서 홀로 지내지만, 가끔 찾아오는 자녀와 손주들을 위해 서툰 글씨로 꾹꾹 눌러 쓴 편지와 간식을 준비하며 그리움을 달래요. 그리고 오시자 씨에게는 꿈이 하나 있어요. 낡은 수첩에 조금씩 적어 온 자신의 인생 이야기를 한 권의 작은 책으로 완성하는 것이에요. 🥹
